\begin{frame}{O que você precisa entregar}
  \begin{itemize}
    \item Um \textbf{artigo científico} em PDF (não é um relatório de estágio).
    \item \textbf{5 a 12 páginas}, contando da Introdução até o final das
          Referências.
    \begin{itemize}
      \item O Resumo não entra nessa contagem.
    \end{itemize}
    \item Formato ABNT: papel A4, fonte Times New Roman ou Arial 12,
          espaçamento 1,5.
  \end{itemize}
\end{frame}

      \textbf{5} & Conclusão \\

% ----------------------------------------------------------------------
\begin{frame}{O que cada seção precisa ter}
  \begin{itemize}
    \item \textbf{Introdução}: contexto, o problema que você investigou,
          objetivo geral (e específicos, se houver).
    \item \textbf{Fundamentação Teórica}: os conceitos técnicos que
          embasam o trabalho, com citação de autores (não é só copiar e
          colar).
    \item \textbf{Metodologia}: o que você fez, com quê, como e quando ---
          detalhado o bastante para alguém repetir.
  \end{itemize}
\end{frame}

\begin{frame}{O que cada seção precisa ter (cont.)}
  \begin{itemize}
    \item \textbf{Resultados e Discussão}: o que você encontrou, e o que
          isso significa --- não é só ``colar números''.
    \item \textbf{Conclusão}: retoma o objetivo, resume o que foi
          encontrado, diz a contribuição do trabalho. Sem dado novo aqui.
    \item \textbf{Referências}: tudo que você citou no texto, e só o que
          você citou.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------------
\begin{frame}{Como funciona a Mostra Tecnológica}
  \begin{itemize}
    \item Você apresenta o trabalho publicamente, para visitantes externos
    \item Professores escolhidos pelo avaliador visitam e avaliam seu trabalho.
    \item Nota de 0 a 100 --- aprovação com, no mínimo, 60 pontos.
    \item Apresentar na Mostra \textbf{já cumpre} a etapa de banca da
          prática profissional.
    \item Orientador avalia texto final e entrega nota final
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------------
\begin{frame}{Quem não puder apresentar na Mostra}
  \begin{alertblock}{Banca à parte}
    Quem não apresentar no dia da Mostra Tecnológica (02/12) terá uma
    \textbf{banca marcada separadamente} para avaliar o artigo.
  \end{alertblock}
  \vfill
  Procure a coordenação ou seu(sua) orientador(a) com antecedência se
  souber que não vai conseguir participar da Mostra. Trabalharemos no agendamento.
\end{frame}

% ----------------------------------------------------------------------
\begin{frame}{Onde conseguir o modelo}
  \begin{itemize}
    \item A coordenação tem um \textbf{modelo de artigo} pronto, em Word e
          em LaTeX:
    \begin{itemize}
      \item uma versão \textbf{explicada}, com instruções em cada seção;
      \item uma versão com um \textbf{exemplo fictício} preenchido, para
            você ver como o artigo pronto deve ficar.
    \end{itemize}
    \item Peça o modelo ao(à) seu(sua) orientador(a) ou diretamente à
          coordenação de curso.
  \end{itemize}
  Um link para o repositório será disponibilizado por e-mail/GSA.
\end{frame}
